# LangGraph + OmicHub 部署与迁移指南

> 版本: v1.0  
> 前置条件: OmicHub 已正常运行，PostgreSQL 和 Redis 可用

## 一、快速开始（推荐）

### 1.1 一键安装

```bash
cd /path/to/omichub

# 下载集成包并解压
wget https://github.com/your-org/omichub-langgraph/archive/main.tar.gz
tar -xzf main.tar.gz
cp -r omichub-langgraph-main/langgraph_integration .

# 运行安装脚本
python langgraph_integration/scripts/install_langgraph.py
```

#### 步骤 5: 更新 main.py

在 `lifespan` 函数中添加 LangGraph 初始化：

```python
from omichub.infrastructure.execution.langgraph_runtime import LangGraphRuntimeService
from omichub.infrastructure.execution.langgraph_nodes import NodeDeps
from omichub.infrastructure.execution.langgraph_executor import register_langgraph_executor
from omichub.infrastructure.checkpoint.postgres_checkpoint import OmicHubCheckpointSaver
from omichub.application.services.hitl_service import HITLService
from omichub.api.v1.agent import set_langgraph_services

@asynccontextmanager
async def lifespan(app: FastAPI):
    # ... 现有初始化代码 ...

    # ── LangGraph 初始化 ──
    try:
        node_deps = NodeDeps(
            agent_service=app.state.agent_service,
            provider_manager=app.state.provider_manager,
            mcp_client=app.state.mcp_client,
            task_service=app.state.task_service,
            chat_service=app.state.chat_service,
        )
        checkpoint_saver = OmicHubCheckpointSaver()
        langgraph_runtime = LangGraphRuntimeService(
            node_deps=node_deps,
            checkpoint_saver=checkpoint_saver,
        )
        await langgraph_runtime.initialize()

        hitl_service = HITLService(langgraph_runtime)

        # 注册到执行器注册表
        from omichub.infrastructure.execution.registry import _EXECUTORS
        _EXECUTORS["langgraph"] = LangGraphExecutor(langgraph_runtime)

        # 注入到 API 路由
        set_langgraph_services(langgraph_runtime, hitl_service)

        app.state.langgraph_runtime = langgraph_runtime
        app.state.hitl_service = hitl_service

        logger.info("LangGraph 运行时初始化完成")
    except Exception as e:
        logger.warning(f"LangGraph 初始化失败: {e}")

    yield

    # ... 关闭代码 ...
    if hasattr(app.state, 'langgraph_runtime'):
        await app.state.langgraph_runtime.shutdown()
```

## 二、配置说明

### 2.1 流程 YAML 配置

在 `flows/*.yaml` 中启用 LangGraph：

```yaml
# flows/rna_seq.yaml
meta:
  id: rna_seq
  name: "RNA-seq 分析"
  version: "2.0.0"
  description: "RNA-seq 差异表达分析流程"

execution:
  engine: langgraph              # 启用 LangGraph（默认 snakemake）
  snakefile: "/workflow/Snakefile"
  container_image: "omichub/flow-rnaseq:v1.0"
  default_resources:
    cores: 8
    memory: "16G"

  # LangGraph 特有配置
  langgraph:
    max_rounds: 8                # 最大对话轮次
    hitl_enabled: true           # 启用 HITL
    hitl_timeout: 600            # HITL 超时秒数
    enable_checkpoint: true      # 启用检查点持久化
    checkpoint_ttl_days: 7       # 检查点保留天数

parameters:
  # ... 现有参数配置 ...
```

## 三、前端集成

### 3.1 AgentWorkspace.vue 改造

```vue
<script setup lang="ts">
import { ref, onMounted } from 'vue'
import { useLangGraphStream } from '@/composables/useLangGraphStream'
import HITLDialog from '@/components/HITLDialog.vue'

const { state, startStream, resumeHITL, cancelStream, fetchPendingHITL } = useLangGraphStream()

const showHITLDialog = ref(false)
const messages = ref<Array<{role: string, content: string}>>([])

// HITL 处理
async function onHITL(request: any) {
  showHITLDialog.value = true
}

async function onHITLConfirm(payload: { requestId: string, comment?: string }) {
  await resumeHITL(payload.requestId, 'confirm', {}, payload.comment)
  showHITLDialog.value = false
}

async function onHITLModify(payload: { requestId: string, modifications: any, comment?: string }) {
  await resumeHITL(payload.requestId, 'modify', payload.modifications, payload.comment)
  showHITLDialog.value = false
}

async function onHITLReject(payload: { requestId: string, comment?: string }) {
  await resumeHITL(payload.requestId, 'reject', {}, payload.comment)
  showHITLDialog.value = false
}

// 发送消息
async function sendMessage(text: string) {
  messages.value.push({ role: 'user', content: text })

  await startStream({
    agentId: 'bio_assistant',
    sessionId: 'current_session',
    message: text,
    onText: (text) => {
      // 打字机效果
      const lastMsg = messages.value[messages.value.length - 1]
      if (lastMsg.role === 'assistant') {
        lastMsg.content += text
      } else {
        messages.value.push({ role: 'assistant', content: text })
      }
    },
    onHITL: onHITL,
    onToolCall: (name, input) => {
      messages.value.push({ role: 'system', content: `🔧 调用工具: ${name}` })
    },
    onToolResult: (name, result, success) => {
      messages.value.push({
        role: 'system',
        content: success ? `✅ ${name} 完成` : `❌ ${name} 失败`
      })
    },
    onDone: (response, metadata) => {
      console.log('完成:', metadata)
    },
    onError: (msg) => {
      messages.value.push({ role: 'error', content: msg })
    },
  })
}

// 页面加载时检查待处理 HITL
onMounted(async () => {
  const pending = await fetchPendingHITL('current_session')
  if (pending.length > 0) {
    onHITL(pending[0])
  }
})
</script>

<template>
  <div class="chat-container">
    <div class="messages">
      <div v-for="(msg, i) in messages" :key="i" :class="`msg-${msg.role}`">
        {{ msg.content }}
      </div>
    </div>

    <div class="input-area">
      <input
        v-model="inputText"
        @keydown.enter="sendMessage(inputText)"
        :disabled="state.isStreaming"
        placeholder="输入消息..."
      />
      <button @click="sendMessage(inputText)" :disabled="state.isStreaming">
        {{ state.isStreaming ? '发送中...' : '发送' }}
      </button>
      <button v-if="state.isStreaming" @click="cancelStream">取消</button>
    </div>

    <!-- HITL 弹窗 -->
    <HITLDialog
      v-model:visible="showHITLDialog"
      :request="state.currentHITLRequest"
      @confirm="onHITLConfirm"
      @modify="onHITLModify"
      @reject="onHITLReject"
    />
  </div>
</template>
```

## 五、回滚方案

如果需要回滚到原有模式：

```bash
# 1. 修改 flows/*.yaml，将 engine: langgraph 改回 engine: snakemake
sed -i 's/engine: langgraph/engine: snakemake/g' flows/*.yaml

# 2. 重启服务（不使用 LangGraph 运行时）
docker-compose restart web

# 3. 可选: 删除 LangGraph 表
# alembic downgrade lg1_langgraph_20260706
```

## 六、故障排查

| 问题 | 可能原因 | 解决方案 |
|------|----------|----------|
| "LangGraph 运行时未初始化" | main.py 未正确初始化 | 检查 lifespan 中的初始化代码 |
| "检查点保存失败" | 数据库表不存在 | 运行 `alembic upgrade head` |
| HITL 弹窗不显示 | 前端未正确监听 hitl_request 事件 | 检查 useLangGraphStream 的 onHITL 回调 |
| 流式输出中断 | SSE 连接超时 | 调整 Nginx proxy_read_timeout |
| 工具调用失败 | MCPClient 未正确注入 | 检查 NodeDeps 的 mcp_client 参数 |
